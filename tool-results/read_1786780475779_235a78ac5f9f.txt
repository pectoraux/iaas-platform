    63→export async function createBuyerProgram(tenantId: string, input: CreateBuyerProgramInput, actorId?: string) {
    64→  const network = await db.networkDefinition.findFirst({ where: { id: input.networkId, tenantId } })
    65→  if (!network) throw new NotFoundError('network', input.networkId)
    66→
    67→  // =========================================================================
    68→  // Version-closed policy boundary (VPP-2C final correctness fix).
    69→  // =========================================================================
    70→  // The architectural invariant is:
    71→  //
    72→  //   Every VPP program must be version-closed:
    73→  //     program.networkVersionId    = N
    74→  //     rewardRule.networkVersionId = N
    75→  //     all runtime policy resolves from N
    76→  //
    77→  // No economic calculation may use policy objects belonging to another
    78→  // version. We resolve the NetworkVersion FIRST, then validate the reward
    79→  // rule belongs to that exact version (and tenant). Both checks are
    80→  // validated lookups — NOT a single transaction — but they run before any
    81→  // write, so a failed validation leaves no partial state.
    82→  // =========================================================================
    83→
    84→  // Step 1: resolve + validate the NetworkVersion.
    85→  let networkVersionId: string
    86→  if (input.networkVersionId) {
    87→    // Explicit version — validate it belongs to this network + tenant and is published.
    88→    // The compound `network: { id, tenantId }` filter enforces both the
    89→    // cross-network and cross-tenant constraints in a single lookup.
    90→    const version = await db.networkVersion.findFirst({
    91→      where: {
    92→        id: input.networkVersionId,
    93→        network: { id: input.networkId, tenantId },
    94→      },
    95→    })
    96→    if (!version) {
    97→      throw new ValidationError(
    98→        `Network version ${input.networkVersionId} does not belong to network ${input.networkId} in this tenant`,
    99→      )
   100→    }
   101→    if (!version.publishedAt) {
   102→      throw new ValidationError(
   103→        `Network version ${input.networkVersionId} (v${version.version}) is not published. ` +
   104→          `A buyer program can only bind to a published, immutable NetworkVersion.`,
   105→      )
   106→    }
   107→    networkVersionId = version.id
   108→  } else {
   109→    // Default to the network's current version — which is always published
   110→    // (publishNetworkVersion sets currentVersionId + publishedAt atomically).
   111→    if (!network.currentVersionId) {
   112→      throw new ValidationError('Network has no published version')
   113→    }
   114→    networkVersionId = network.currentVersionId
   115→  }
   116→
   117→  // Step 2: validate the reward rule belongs to the SAME NetworkVersion.
   118→  // The reward rule directly determines economic settlement — it MUST be
   119→  // version-closed with the program. A V12 program using a V13 reward rule
   120→  // would mix policies from two immutable versions, exactly the split we
   121→  // eliminated for baselines and verification.
   122→  const rule = await db.rewardRule.findFirst({
   123→    where: {
   124→      id: input.rewardRuleId,
   125→      tenantId,
   126→      networkVersionId,
   127→    },
   128→  })
   129→  if (!rule) {
   130→    throw new ValidationError(
   131→      `Reward rule ${input.rewardRuleId} does not belong to network version ${networkVersionId} ` +
   132→        `in this tenant. A buyer program's reward rule must be version-closed with its NetworkVersion.`,
   133→    )
   134→  }
   135→
   136→  // Step 3: for energy_vpp networks, require an accepted baseline policy on
   137→  // the bound version. A version without an accepted baseline policy cannot
   138→  // produce a valid settlement (no persisted strategy → BASELINE_UNAVAILABLE
   139→  // at execution time). Rejecting at program creation surfaces the
   140→  // misconfiguration early, rather than letting every dispatch fail at runtime.
   141→  //
   142→  // This removes the silent hardcoded-default fallback: an old or
   143→  // incompletely configured version can no longer execute using the
   144→  // source-code default 'weekday_weekend_average' baseline.
   145→  if (network.vertical === 'energy_vpp') {
   146→    const policy = await getBaselinePolicyStatus(networkVersionId)
   147→    if (!policy.hasPolicy) {
   148→      throw new ValidationError(
   149→        `Network version ${networkVersionId} has no persisted baseline policy. ` +
   150→          `An energy_vpp buyer program requires a version with an accepted baseline policy ` +
   151→          `(run runAndPersistBaselineEvaluation before binding a program).`,
   152→      )
   153→    }
   154→    if (policy.status !== 'accepted' || !policy.selectedStrategy) {
   155→      throw new ValidationError(
   156→        `Network version ${networkVersionId} baseline policy status is '${policy.status}' ` +
   157→          `(selectedStrategy: ${policy.selectedStrategy ?? 'null'}). ` +
   158→          `An energy_vpp buyer program requires a version with status='accepted'.`,
   159→      )
   160→    }
   161→  }
   162→
   163→  const program = await db.vppBuyerProgram.create({
   164→    data: {
   165→      tenantId,
   166→      networkId: input.networkId,
   167→      networkVersionId,
   168→      name: input.name,
   169→      description: input.description ?? null,
   170→      rewardRuleId: input.rewardRuleId,
   171→      dispatchWindowStart: input.dispatchWindowStart,
   172→      dispatchWindowEnd: input.dispatchWindowEnd,
   173→      pricePerKwh: input.pricePerKwh,
   174→      currency: input.currency ?? 'USD',
   175→      minCapacityKw: input.minCapacityKw,
   176→      maxCapacityKw: input.maxCapacityKw ?? null,
   177→    },
   178→  })
   179→
   180→  await appendAudit({
   181→    tenantId, actorId,
   182→    eventType: 'vpp.program_created',
   183→    resourceType: 'vpp_buyer_program',
   184→    resourceId: program.id,
   185→    metadata: { name: program.name, networkId: input.networkId, networkVersionId },
   186→  })
   187→
   188→  return program
   189→}

   191→/**
   192→ * Inspect a NetworkVersion's persisted baseline policy and return its status.
   193→ * Returns null when no policy is persisted.
   194→ *
   195→ * Used by createBuyerProgram to enforce the strict-baseline rule for
   196→ * energy_vpp networks: a version without an accepted baseline policy cannot
   197→ * host a buyer program.
   198→ */
   199→async function getBaselinePolicyStatus(networkVersionId: string): Promise<{
   200→  hasPolicy: boolean
   201→  status: string | null
   202→  selectedStrategy: string | null
   203→}> {
   204→  const version = await db.networkVersion.findUnique({
   205→    where: { id: networkVersionId },
   206→    select: { baselinePolicyJson: true },
   207→  })
   208→  if (!version?.baselinePolicyJson) {
   209→    return { hasPolicy: false, status: null, selectedStrategy: null }
   210→  }
   211→  const policy = JSON.parse(version.baselinePolicyJson)
   212→  return {
   213→    hasPolicy: true,
   214→    status: policy.status ?? null,
   215→    selectedStrategy: policy.selectedStrategy ?? null,
   216→  }
   217→}

   244→export async function createCapacityReservation(tenantId: string, input: CreateReservationInput, actorId?: string) {
   245→  const program = await db.vppBuyerProgram.findFirst({ where: { id: input.programId, tenantId } })
   246→  if (!program) throw new NotFoundError('vpp_buyer_program', input.programId)
   247→
   248→  // Task 3: validate the full chain.
   249→  // 1. Operator owns the asset.
   250→  const asset = await db.asset.findFirst({ where: { id: input.assetId, tenantId } })
   251→  if (!asset) throw new NotFoundError('asset', input.assetId)
   252→  if (asset.operatorId !== input.operatorId) {
   253→    throw new ValidationError(`Operator ${input.operatorId} does not own asset ${input.assetId}`)
   254→  }
   255→
   256→  // 2. Asset is assigned to the program's network with this capability.
   257→  const assignment = await db.assetNetworkAssignment.findFirst({
   258→    where: {
   259→      tenantId, assetId: input.assetId, networkId: program.networkId,
   260→      capabilityType: input.capabilityType, status: 'active',
   261→    },
   262→  })
   263→  if (!assignment) {
   264→    throw new ValidationError(
   265→      `Asset ${input.assetId} is not assigned to network ${program.networkId} with capability ${input.capabilityType}`,
   266→    )
   267→  }
   268→
   269→  // ATOMIC: VPP reservation + capacity reservation in ONE transaction.
   270→  // The VPP reservation is created first, then the platform capacity
   271→  // reservation with sourceId = vppReservation.id.
   272→  const result = await db.$transaction(async (tx) => {
   273→    // Create the VPP reservation first (inside the transaction).
   274→    const vppReservation = await tx.vppCapacityReservation.create({
   275→      data: {
   276→        tenantId,
   277→        programId: input.programId,
   278→        operatorId: input.operatorId,
   279→        assetId: input.assetId,
   280→        capabilityType: input.capabilityType,
   281→        reservedKw: input.reservedKw,
   282→        reservedKwh: input.reservedKwh ?? null,
   283→        effectiveFrom: new Date(input.startTime),
   284→        effectiveTo: new Date(input.endTime),
   285→      },
   286→    })
   287→
   288→    // Create the platform capacity reservation (inside the same transaction).
   289→    // Physical capacity is resolved from the verified assignment.
   290→    const capacityReservation = await allocateReservation({
   291→      tenantId,
   292→      assetId: input.assetId,
   293→      networkId: program.networkId,
   294→      capabilityType: input.capabilityType,
   295→      requestedAmount: input.reservedKw,
   296→      startTime: new Date(input.startTime),
   297→      endTime: new Date(input.endTime),
   298→      sourceType: 'vpp_reservation',
   299→      sourceId: vppReservation.id,
   300→    }, tx)
   301→
   302→    return { vppReservation, capacityReservation }
   303→  }, { timeout: 30000 })
   304→
   305→  const { vppReservation, capacityReservation } = result
   306→
   307→  await appendAudit({
   308→    tenantId, actorId,
   309→    eventType: 'vpp.reservation_created',
   310→    resourceType: 'vpp_capacity_reservation',
   311→    resourceId: vppReservation.id,
   312→    metadata: {
   313→      programId: input.programId, assetId: input.assetId, capabilityType: input.capabilityType,
   314→      reservedKw: input.reservedKw,
   315→      capacityReservationId: capacityReservation.reservationId,
   316→    },
   317→  })
   318→
   319→  return { reservation: vppReservation, capacityReservation }
   320→}

   326→export interface CreateDispatchInput {
   327→  programId: string
   328→  requestedKw: string
   329→  requestedKwh: string
   330→  startTime: string
   331→  endTime: string
   332→  reason?: string
   333→}
   334→
   335→export async function createDispatch(tenantId: string, input: CreateDispatchInput, actorId?: string) {
   336→  const program = await db.vppBuyerProgram.findFirst({
   337→    where: { id: input.programId, tenantId },
   338→    include: { reservations: { where: { status: 'active' }, include: { asset: true } }, network: true },
   339→  })
   340→  if (!program) throw new NotFoundError('vpp_buyer_program', input.programId)
   341→
   342→  const startTime = new Date(input.startTime)
   343→  const endTime = new Date(input.endTime)
   344→  const requestedKw = new Prisma.Decimal(input.requestedKw)
   345→  const requestedKwh = new Prisma.Decimal(input.requestedKwh)
   346→
   347→  // Task 5: transactional dispatch — create dispatch + allocations + assignments atomically.
   348→  const dispatch = await db.$transaction(async (tx) => {
   349→    // Lock all active reservations for this program.
   350→    await tx.$queryRaw`
   351→      SELECT * FROM "VppCapacityReservation"
   352→      WHERE "programId" = ${input.programId}
   353→        AND status = 'active'
   354→        AND "effectiveFrom" < ${endTime}
   355→        AND "effectiveTo" > ${startTime}
   356→      FOR UPDATE
   357→    `
   358→
   359→    // Reload reservations inside the lock.
   360→    const reservations = await tx.vppCapacityReservation.findMany({
   361→      where: {
   362→        programId: input.programId,
   363→        status: 'active',
   364→        effectiveFrom: { lt: endTime },
   365→        effectiveTo: { gt: startTime },
   366→      },
   367→    })
   368→
   369→    if (reservations.length === 0) {
   370→      throw new ValidationError(`No active reservations for program ${input.programId} in the requested time window`)
   371→    }
   372→
   373→    // Check sufficient reserved capacity.
   374→    const totalReservedKw = reservations.reduce(
   375→      (sum, r) => sum.plus(new Prisma.Decimal(r.reservedKw)),
   376→      new Prisma.Decimal(0),
   377→    )
   378→    if (totalReservedKw.lessThan(requestedKw)) {
   379→      throw new ValidationError(
   380→        `Insufficient reserved capacity: ${totalReservedKw.toString()} kW < requested ${requestedKw.toString()} kW`,
   381→      )
   382→    }
   383→
   384→    // Create the dispatch.
   385→    const created = await tx.vppDispatch.create({
   386→      data: {
   387→        tenantId,
   388→        programId: input.programId,
   389→        requestedKw: input.requestedKw,
   390→        requestedKwh: input.requestedKwh,
   391→        startTime,
   392→        endTime,
   393→        reason: input.reason ?? null,
   394→        status: 'assigned',
   395→      },
   396→    })
   397→
   398→    // Assign assets proportionally + create capacity commitments (atomic).
   399→    // Fix 1: each assignment creates its OWN CapacityCommitment (1:1).
   400→    // The commitmentId is stored on the assignment for direct lookup.
   401→    for (const reservation of reservations) {
   402→      const ratio = new Prisma.Decimal(reservation.reservedKw).div(totalReservedKw)
   403→      const assignedKw = requestedKw.times(ratio)
   404→      const assignedKwh = requestedKwh.times(ratio)
   405→
   406→      // Find the platform capacity reservation for this VPP reservation (inside tx).
   407→      const capacityReservation = await tx.capacityReservation.findFirst({
   408→        where: { tenantId, sourceType: 'vpp_reservation', sourceId: reservation.id, status: 'active' },
   409→      })
   410→
   411→      let commitmentId: string | null = null
   412→      if (capacityReservation) {
   413→        // Create a capacity commitment for THIS assignment (not the dispatch).
   414→        const commitment = await createCapacityCommitment({
   415→          tenantId,
   416→          reservationId: capacityReservation.id,
   417→          committedAmount: assignedKw.toFixed(8),
   418→          unit: 'kW',
   419→          startTime,
   420→          endTime,
   421→          sourceType: 'vpp_dispatch_assignment',
   422→          sourceId: `assignment-${created.id}-${reservation.assetId}`, // unique per assignment
   423→        }, tx)
   424→        commitmentId = commitment.commitmentId
   425→      }
   426→
   427→      // Store the commitmentId on the assignment for direct lookup.
   428→      await tx.vppDispatchAssignment.create({
   429→        data: {
   430→          tenantId,
   431→          dispatchId: created.id,
   432→          assetId: reservation.assetId,
   433→          operatorId: reservation.operatorId,
   434→          capabilityType: reservation.capabilityType,
   435→          assignedKw: assignedKw.toFixed(8),
   436→          assignedKwh: assignedKwh.toFixed(8),
   437→          capacityCommitmentId: commitmentId,
   438→        },
   439→      })
   440→    }
   441→
   442→    return created
   443→  }, { timeout: 30000 })
   444→
   445→  await appendAudit({
   446→    tenantId, actorId,
   447→    eventType: 'vpp.dispatch_created',
   448→    resourceType: 'vpp_dispatch',
   449→    resourceId: dispatch.id,
   450→    metadata: { programId: input.programId, requestedKw: input.requestedKw },
   451→  })
   452→
   453→  const assignments = await db.vppDispatchAssignment.findMany({ where: { dispatchId: dispatch.id } })
   454→  return { dispatch, assignments }
   455→}

   457→// ---------------------------------------------------------------------------
   458→// Dispatch Execution (task 7: idempotent, task 6: no auto-funding)
   459→// ---------------------------------------------------------------------------
   460→
   461→/**
   462→ * Execute a dispatch assignment using the DER adapter.
   463→ *
   464→ * Task 7: IDEMPOTENT — if the assignment is already completed, returns the
   465→ * existing result. No duplicate telemetry, attestations, contributions, or rewards.
   466→ *
   467→ * Task 6: NO AUTO-FUNDING — if buyer funds are insufficient, the reward
   468→ * posting fails (the assignment remains in 'dispatching' state). The buyer
   469→ * must be pre-funded.
   470→ */
   471→export async function executeDispatchAssignment(
   472→  tenantId: string,
   473→  assignmentId: string,
   474→  provisioningSecret: string,
   475→  actorId?: string,
   476→) {
   477→  const assignment = await db.vppDispatchAssignment.findFirst({
   478→    where: { id: assignmentId, tenantId },
   479→    include: {
   480→      asset: { include: { devices: { include: { credential: true } } } },
   481→      dispatch: { include: { program: { include: { network: true, networkVersion: true } } } },
   482→    },
   483→  })
   484→  if (!assignment) throw new NotFoundError('vpp_dispatch_assignment', assignmentId)
   485→
   486→  // Task 7: idempotency — if already completed, return existing result.
   487→  if (assignment.status === 'completed') {
   488→    // Fetch the reward + settlement linked to this assignment's contribution.
   489→    const reward = assignment.contributionId
   490→      ? await db.reward.findFirst({ where: { contributionId: assignment.contributionId } })
   491→      : null
   492→    const settlement = reward
   493→      ? await db.settlement.findUnique({ where: { rewardId: reward.id } })
   494→      : null
   495→    return {
   496→      assignment_id: assignmentId,
   497→      event_id: assignment.eventId,
   498→      contribution_id: assignment.contributionId,
   499→      reward_id: reward?.id ?? null,
   500→      settlement_id: settlement?.id ?? null,
   501→      performance_kwh: assignment.performanceKwh,
   502→      actual_kwh: assignment.actualKwh,
   503→      baseline_kwh: assignment.baselineKwh,
   504→      duplicate: true,
   505→      message: 'Assignment already completed',
   506→    }
   507→  }
   508→
   509→  // Fix 2: atomic status transition — ONLY 'assigned' can transition to 'dispatching'.
   510→  const updated = await db.vppDispatchAssignment.updateMany({
   511→    where: { id: assignmentId, status: 'assigned' },
   512→    data: { status: 'dispatching' },
   513→  })
   514→  if (updated.count === 0) {
   515→    throw new ConflictError(`Assignment ${assignmentId} is not in 'assigned' state (current: ${assignment.status}). Concurrent execution rejected.`)
   516→  }
   517→
   518→  // State machine:
   519→  //   ASSIGNED → DISPATCHING → DELIVERY_VERIFIED → USAGE_RECORDED → SETTLEMENT_PENDING → COMPLETED
   520→  //   Pre-usage failure: → FAILED → commitment RELEASED (no money moved)
   521→  //   Post-usage failure: → RECONCILIATION_REQUIRED → commitment stays CONSUMED (liability exists)
   522→  const releaseAssignmentCapacity = async () => {
   523→    if (assignment.capacityCommitmentId) {
   524→      await releaseCommitment(tenantId, 'vpp_dispatch_assignment', `assignment-${assignment.dispatchId}-${assignment.assetId}`)
   525→    }
   526→  }
   527→
   528→  // Pre-usage failure: release capacity (no irreversible action has occurred).
   529→  const failAssignment = async () => {
   530→    await db.vppDispatchAssignment.update({ where: { id: assignmentId }, data: { status: 'failed' } })
   531→    await releaseAssignmentCapacity()
   532→  }
   533→
   534→  // Post-usage failure: capacity is CONSUMED, money may have moved.
   535→  // Do NOT release. Enter reconciliation state for retry.
   536→  const markReconciliationRequired = async (reason: string) => {
   537→    await db.vppDispatchAssignment.update({
   538→      where: { id: assignmentId },
   539→      data: { status: 'reconciliation_required' },
   540→    })
   541→    await appendAudit({
   542→      tenantId, actorId,
   543→      eventType: 'vpp.reconciliation_required',
   544→      resourceType: 'vpp_dispatch_assignment',
   545→      resourceId: assignmentId,
   546→      metadata: { reason, commitmentId: assignment.capacityCommitmentId },
   547→    })
   548→  }
   549→
   550→  // Track whether usage has been recorded (determines failure handling).
   551→  let usageRecorded = false
   552→
   553→  try {
   554→    const device = assignment.asset.devices.find((d) => d.credential && d.credential.status === 'active')
   555→    if (!device) {
   556→      throw new ValidationError(`Asset ${assignment.assetId} has no active device with credential`)
   557→    }
   558→
   559→    // --- DER adapter (can throw on network/hardware errors) ---
   560→    const durationSeconds = Math.floor(
   561→      (assignment.dispatch.endTime.getTime() - assignment.dispatch.startTime.getTime()) / 1000,
   562→    )
   563→    const dischargeResult = await derAdapter.executeDischarge({
   564→      assignedKw: assignment.assignedKw,
   565→      assignedKwh: assignment.assignedKwh,
   566→      capabilityType: assignment.capabilityType,
   567→      durationSeconds,
   568→    })
   569→
   570→    // Sign + submit telemetry as a generic Event.
   571→    const eventId = `vpp-dispatch-${assignmentId}-${Date.now()}`
   572→    const timestamp = new Date().toISOString()
   573→    const sequence = Math.floor(Date.now() / 1000)
   574→    const message = buildCanonicalMessage({
   575→      device_id: device.id,
   576→      event_id: eventId,
   577→      timestamp,
   578→      event_type: 'telemetry',
   579→      sequence,
   580→      payload: dischargeResult.telemetry.payload,
   581→    })
   582→    const signingKey = deriveSigningKey(provisioningSecret)
   583→    const signature = signMessage(message, signingKey)
   584→
   585→    // =========================================================================
   586→    // IMMUTABLE POLICY BOUNDARY (VPP-2C final correctness fix).
   587→    // =========================================================================
   588→    // Every event AND every economic calculation associated with this dispatch
   589→    // MUST resolve against the SAME immutable NetworkVersion — the one the
   590→    // buyer program was bound to at creation time
   591→    // (assignment.dispatch.program.networkVersionId).
   592→    //
   593→    // Previously, telemetry ingestion resolved `latest published version for
   594→    // network.id` at execution time. That produced a version split:
   595→    //   baseline policy = V12 (program.networkVersionId)
   596→    //   verification    = V13 (network.currentVersionId)
   597→    // which is exactly the historical-reproducibility violation we eliminated
   598→    // for baselines. The same rule now applies to verification, reward rules,
   599→    // and contribution policy: all reference the program's bound version.
   600→    //
   601→    // The programVersion is already eager-loaded on the assignment query
   602→    // (dispatch.program.networkVersion), so no extra DB round-trip is needed.
   603→    // =========================================================================
   604→    const programVersion = assignment.dispatch.program.networkVersion
   605→    if (!programVersion) {
   606→      throw new Error('BASELINE_UNAVAILABLE: program has no bound network version')
   607→    }
   608→    if (!programVersion.publishedAt) {
   609→      // A program should never be bound to an unpublished version
   610→      // (createBuyerProgram enforces this), but defend in depth.
   611→      throw new Error(
   612→        `BASELINE_UNAVAILABLE: program's network version ${programVersion.id} (v${programVersion.version}) is not published`,
   613→      )
   614→    }
   615→
   616→    const ingestResult = await ingestEvent(tenantId, {
   617→      device_id: device.id,
   618→      event_id: eventId,
   619→      timestamp,
   620→      event_type: 'telemetry',
   621→      sequence,
   622→      payload: dischargeResult.telemetry.payload,
   623→      signature,
   624→      // SAME immutable version used for baseline/reward/contribution below.
   625→      network_version_id: programVersion.id,
   626→      capability_type: assignment.capabilityType,
   627→    })
   628→
   629→    await processEventOutbox(tenantId)
   630→
   631→    const event = await db.event.findUnique({
   632→      where: { id: ingestResult.event_id },
   633→      include: { attestations: true },
   634→    })
   635→
   636→    if (event?.status !== 'verified' || !event.attestations[0]) {
   637→      throw new Error(`Dispatch telemetry verification failed: ${event?.status}`)
   638→    }
   639→
   640→    const attestation = event.attestations[0]
   641→
   642→    // --- Baseline calculation (VPP-2C: production baseline via HistoricalTelemetryProvider) ---
   643→    // FIX: Uses BaselineContext (production input) — NEVER ground truth.
   644→    // FIX: Resolves strategy from persisted policy (not hardcoded).
   645→    // FIX: Prevents negative performance payments: max(0, actual - baseline).
   646→    const actualKwh = new Prisma.Decimal(dischargeResult.actualKwh)
   647→
   648→    const { SimulatedHistoricalTelemetryProvider } = await import('./historical-telemetry-provider.service')
   649→    const baselineEngine = await import('./baseline-engine.service')
   650→    const getStrategy = baselineEngine.getStrategy
   651→    type BaselineContext = baselineEngine.BaselineContext
   652→    const telemetryProvider = new SimulatedHistoricalTelemetryProvider()
   653→
   654→    // Get historical telemetry (training data — strictly before dispatch).
   655→    const historicalDays = await telemetryProvider.getHistory(
   656→      assignment.assetId,
   657→      assignment.dispatch.startTime,
   658→      14,
   659→    )
   660→
   661→    if (!historicalDays || historicalDays.length < 3) {
   662→      throw new Error('BASELINE_UNAVAILABLE: insufficient historical telemetry for baseline calculation')
   663→    }
   664→
   665→    // Build BaselineContext from observable dispatch parameters ONLY.
   666→    // This contains NO ground truth (no trueCounterfactual, no trueIncremental).
   667→    const dispatchDurationHours = Math.ceil(
   668→      (assignment.dispatch.endTime.getTime() - assignment.dispatch.startTime.getTime()) / 3600000,
   669→    )
   670→    const dispatchHour = assignment.dispatch.startTime.getHours()
   671→    const dispatchDate = assignment.dispatch.startTime.toISOString().split('T')[0]
   672→    const dayOfWeek = assignment.dispatch.startTime.getDay()
   673→
   674→    const baselineContext: BaselineContext = {
   675→      dispatchStartIndex: dispatchHour * 4,
   676→      dispatchEndIndex: Math.min(95, dispatchHour * 4 + dispatchDurationHours * 4),
   677→      dispatchDate,
   678→      dayOfWeek,
   679→      isWeekend: dayOfWeek === 0 || dayOfWeek === 6,
   680→    }
   681→
   682→    // Resolve the baseline strategy from the PERSISTED NetworkVersion policy.
   683→    // CRITICAL: uses the SAME programVersion object we resolved above for
   684→    // telemetry verification. This guarantees baseline policy, verification
   685→    // policy, reward rules, and contribution policy all reference the exact
   686→    // same immutable NetworkVersion — no version split is possible.
   687→    //
   688→    // A dispatch created under V12 always uses V12's policy, even after V13
   689→    // is published and becomes current.
   690→    //
   691→    // STRICT POLICY (VPP-2C final correctness fix): there is NO hardcoded
   692→    // fallback. A version without an accepted persisted baseline policy
   693→    // cannot execute. createBuyerProgram enforces this for new programs; this
   694→    // block is the defend-in-depth check for any program that pre-dates the
   695→    // strict rule or was created via direct DB access. Such a program gets
   696→    // BASELINE_UNAVAILABLE and enters RECONCILIATION_REQUIRED — it never
   697→    // silently uses a source-code default baseline.
   698→    const versionForPolicy = programVersion
   699→
   700→    if (!versionForPolicy.baselinePolicyJson) {
   701→      throw new Error(
   702→        `BASELINE_UNAVAILABLE: network version ${versionForPolicy.id} (v${versionForPolicy.version}) ` +
   703→          `has no persisted baseline policy. No hardcoded fallback is permitted.`,
   704→      )
   705→    }
   706→    const policy = JSON.parse(versionForPolicy.baselinePolicyJson)
   707→    if (policy.status !== 'accepted' || !policy.selectedStrategy) {
   708→      throw new Error(
   709→        `BASELINE_UNAVAILABLE: network version ${versionForPolicy.id} (v${versionForPolicy.version}) ` +
   710→          `baseline policy status is '${policy.status}' (no accepted strategy).`,
   711→      )
   712→    }
   713→    const strategyName: string = policy.selectedStrategy
   714→
   715→    const baselineStrategy = getStrategy(strategyName)
   716→    if (!baselineStrategy) {
   717→      throw new Error(`BASELINE_UNAVAILABLE: strategy '${strategyName}' not found in registry`)
   718→    }
   719→
   720→    const baselinePrediction = baselineStrategy.predict(historicalDays, baselineContext)
   721→
   722→    const baselineKwh = new Prisma.Decimal(baselinePrediction.predictedCounterfactualKwh)
   723→    // FIX: Prevent negative performance payments.
   724→    // rawPerformanceKwh preserves the signed value for analytics.
   725→    // verifiedPerformanceKwh is max(0, actual - baseline) — the economically payable quantity.
   726→    const rawPerformanceKwh = actualKwh.minus(baselineKwh)
   727→    const verifiedPerformanceKwh = rawPerformanceKwh.isNegative()
   728→      ? new Prisma.Decimal(0)
   729→      : rawPerformanceKwh
   730→
   731→    // Fetch ground truth for METADATA ONLY (not used in baseline calculation).
   732→    const groundTruth = await telemetryProvider.getDispatchDayGroundTruth?.(
   733→      assignment.assetId,
   734→      assignment.dispatch.startTime,
   735→      dispatchDurationHours,
   736→      parseFloat(assignment.assignedKw),
   737→    )
   738→
   739→    const baseline = await db.vppBaseline.create({
   740→      data: {
   741→        tenantId,
   742→        assignmentId,
   743→        method: baselinePrediction.method,
   744→        baselineKw: '0',
   745→        baselineKwh: baselineKwh.toString(),
   746→        actualKw: dischargeResult.actualKw,
   747→        actualKwh: actualKwh.toString(),
   748→        performanceKwh: verifiedPerformanceKwh.toString(),
   749→        metadataJson: JSON.stringify({
   750→          attestationId: attestation.id,
   751→          baselineMethod: baselinePrediction.method,
   752→          strategyName,
   753→          // Audit trail: the EXACT NetworkVersion this baseline was computed
   754→          // against. Must equal event.networkVersionId — if they ever diverge
   755→          // it indicates a version-split bug.
   756→          networkVersionId: programVersion.id,
   757→          networkVersionNumber: programVersion.version,
   758→          predictedCounterfactualKwh: baselinePrediction.predictedCounterfactualKwh,
   759→          rawPerformanceKwh: rawPerformanceKwh.toString(), // signed value for analytics
   760→          verifiedPerformanceKwh: verifiedPerformanceKwh.toString(), // non-negative payable
   761→          negativePerformanceClipped: rawPerformanceKwh.isNegative(),
   762→          trueCounterfactualKwh: groundTruth?.trueCounterfactualKwh ?? null, // metadata only
   763→          trueIncrementalKwh: groundTruth?.trueIncrementalKwh ?? null, // metadata only
   764→          historyDays: historicalDays.length,
   765→          provider: 'simulated',
   766→        }),
   767→      },
   768→    })
   769→
   770→    // --- Derived contribution (uses verifiedPerformanceKwh — never negative) ---
   771→    const contribution = await createContribution(
   772→      tenantId,
   773→      {
   774→        attestationIds: [attestation.id],
   775→        derivedQuantity: verifiedPerformanceKwh.toString(),
   776→        derivedUnit: 'kWh',
   777→      },
   778→      `vpp-baseline-${baseline.id}`,
   779→    )
   780→
   781→    // Persist delivery results + stage = delivery_verified.
   782→    await db.vppDispatchAssignment.update({
   783→      where: { id: assignmentId },
   784→      data: {
   785→        economicStage: 'delivery_verified',
   786→        actualKwh: actualKwh.toString(),
   787→        baselineKwh: baselineKwh.toString(),
   788→        performanceKwh: verifiedPerformanceKwh.toString(),
   789→        eventId: event.id,
   790→        contributionId: contribution.id,
   791→      },
   792→    })
   793→
   794→    // --- RECORD USAGE BEFORE ANY IRREVERSIBLE FINANCIAL SETTLEMENT ---
   795→    if (assignment.capacityCommitmentId) {
   796→      await recordUsage({
   797→        tenantId,
   798→        commitmentId: assignment.capacityCommitmentId,
   799→        quantity: actualKwh.toString(),
   800→        unit: 'kWh',
   801→        startTime: assignment.dispatch.startTime,
   802→        endTime: assignment.dispatch.endTime,
   803→        sourceType: 'vpp_dispatch_assignment',
   804→        sourceId: `assignment-${assignment.dispatchId}-${assignment.assetId}`,
   805→      })
   806→    }
   807→    usageRecorded = true
   808→    await db.vppDispatchAssignment.update({ where: { id: assignmentId }, data: { economicStage: 'usage_recorded' } })
   809→
   810→    // --- Reward ---
   811→    const reward = await calculateReward(tenantId, contribution.id, `vpp-contrib-${contribution.id}`)
   812→    await db.vppDispatchAssignment.update({ where: { id: assignmentId }, data: { economicStage: 'reward_calculated' } })
   813→
   814→    // --- Ledger ---
   815→    await postRewardToLedger(tenantId, { rewardId: reward.id }, `vpp-reward-${reward.id}`)
   816→    await db.vppDispatchAssignment.update({ where: { id: assignmentId }, data: { economicStage: 'ledger_posted' } })
   817→
   818→    // --- Settlement ---
   819→    const settlement = await createSettlement(tenantId, reward.id)
   820→    await db.vppDispatchAssignment.update({ where: { id: assignmentId }, data: { economicStage: 'settlement_pending' } })
   821→
   822→    // Use the TARGETED, LEASE-SAFE settlement processor — NOT the tenant-wide outbox.
   823→    // This ensures: settlement status is the source of truth for assignment completion.
   824→    // If settlement fails → assignment enters RECONCILIATION_REQUIRED (not COMPLETED).
   825→    const settlementResult = await processSettlementForReward(tenantId, reward.id)
   826→    if (!settlementResult.completed) {
   827→      // Settlement did not complete. The assignment must NOT be marked COMPLETED.
   828→      // Enter reconciliation state — capacity stays consumed, usage stays,
   829→      // financial liability exists. Reconciliation can retry.
   830→      throw new Error(`Settlement not completed (status: ${settlementResult.settlementId})`)
   831→    }
   832→
   833→    // --- COMPLETED (only reached if settlement succeeded) ---
   834→    await db.vppDispatchAssignment.update({
   835→      where: { id: assignmentId },
   836→      data: { status: 'completed', economicStage: 'completed', completedAt: new Date() },
   837→    })
   838→
   839→    // Update dispatch status if all assignments completed.
   840→    const pendingAssignments = await db.vppDispatchAssignment.count({
   841→      where: { dispatchId: assignment.dispatchId, status: { not: 'completed' } },
   842→    })
   843→    if (pendingAssignments === 0) {
   844→      await db.vppDispatch.update({ where: { id: assignment.dispatchId }, data: { status: 'completed' } })
   845→    }
   846→
   847→    await appendAudit({
   848→      tenantId, actorId,
   849→      eventType: 'vpp.dispatch_completed',
   850→      resourceType: 'vpp_dispatch_assignment',
   851→      resourceId: assignmentId,
   852→      metadata: {
   853→        actualKwh: actualKwh.toString(),
   854→        performanceKwh: verifiedPerformanceKwh.toString(),
   855→        eventId: event.id,
   856→        contributionId: contribution.id,
   857→        rewardId: reward.id,
   858→        contributionQuantity: contribution.quantity,
   859→      },
   860→    })
   861→
   862→    return {
   863→      assignment_id: assignmentId,
   864→      event_id: event.id,
   865→      attestation_id: attestation.id,
   866→      baseline_id: baseline.id,
   867→      contribution_id: contribution.id,
   868→      reward_id: reward.id,
   869→      settlement_id: settlement.id,
   870→      performance_kwh: verifiedPerformanceKwh.toString(),
   871→      actual_kwh: actualKwh.toString(),
   872→      baseline_kwh: baselineKwh.toString(),
   873→      contribution_quantity: contribution.quantity,
   874→      duplicate: false,
   875→    }
   876→  } catch (err) {
   877→    // STATE MACHINE FAILURE HANDLING:
   878→    // Pre-usage failure (usageRecorded = false): → FAILED → release capacity.
   879→    //   No irreversible action has occurred. Safe to release.
   880→    // Post-usage failure (usageRecorded = true): → RECONCILIATION_REQUIRED.
   881→    //   Capacity is CONSUMED. Money may have moved. Do NOT release.
   882→    //   Financial reconciliation can retry settlement.
   883→    if (usageRecorded) {
   884→      await markReconciliationRequired(err instanceof Error ? err.message : 'Post-usage failure')
   885→    } else {
   886→      await failAssignment()
   887→    }
   888→    throw err
   889→  }
   890→}
   891→
   892→// ---------------------------------------------------------------------------
   893→// Reconciliation (RECONCILIATION_REQUIRED → RECONCILING → COMPLETED)
   894→// ---------------------------------------------------------------------------
   895→
   896→/**
   897→ * Reconcile an assignment in RECONCILIATION_REQUIRED state.
   898→ *
   899→ * Inspects durable state (economicStage + linked entities) and resumes at the
   900→ * first missing economic stage:
   901→ *
   902→ *   usage exists, reward missing → calculate reward
   903→ *   reward exists, ledger missing → post to ledger
   904→ *   ledger exists, settlement missing → create settlement
   905→ *   settlement exists but not completed → process settlement outbox
   906→ *   settlement completed → mark assignment COMPLETED
   907→ *
   908→ * Every stage is idempotent. Does NOT re-execute DER dispatch or create
   909→ * duplicate usage.
   910→ *
   911→ * Concurrency-safe: RECONCILIATION_REQUIRED → RECONCILING atomic claim.
   912→ */
   913→export async function reconcileAssignment(
   914→  tenantId: string,
   915→  assignmentId: string,
   916→  actorId?: string,
   917→): Promise<{ assignment_id: string; status: string; economic_stage: string; message: string }> {
   918→  // Atomic claim: only one reconciler can proceed.
   919→  const claimed = await db.vppDispatchAssignment.updateMany({
   920→    where: { id: assignmentId, tenantId, status: 'reconciliation_required' },
   921→    data: { status: 'reconciling' },
   922→  })
   923→  if (claimed.count === 0) {
   924→    const current = await db.vppDispatchAssignment.findFirst({ where: { id: assignmentId, tenantId } })
   925→    return {
   926→      assignment_id: assignmentId,
   927→      status: current?.status ?? 'unknown',
   928→      economic_stage: current?.economicStage ?? 'unknown',
   929→      message: `Assignment is not in reconciliation_required state (current: ${current?.status})`,
   930→    }
   931→  }
   932→
   933→  const assignment = await db.vppDispatchAssignment.findFirst({
   934→    where: { id: assignmentId, tenantId },
   935→    include: { dispatch: true },
   936→  })
   937→  if (!assignment) throw new NotFoundError('vpp_dispatch_assignment', assignmentId)
   938→
   939→  try {
   940→    // Inspect DURABLE OBJECTS (not just economicStage) to determine the actual
   941→    // next stage. The economicStage checkpoint is a hint, but the existence/state
   942→    // of reward, ledger posting, and settlement is the source of truth.
   943→    if (!assignment.contributionId) {
   944→      throw new Error('Cannot reconcile: contributionId missing')
   945→    }
   946→
   947→    // 1. Check if reward exists. If not, calculate it.
   948→    let reward = await db.reward.findFirst({ where: { contributionId: assignment.contributionId } })
   949→    if (!reward) {
   950→      reward = await calculateReward(tenantId, assignment.contributionId, `vpp-contrib-${assignment.contributionId}`)
   951→      await db.vppDispatchAssignment.update({ where: { id: assignmentId }, data: { economicStage: 'reward_calculated' } })
   952→    }
   953→
   954→    // 2. Check if reward is posted to ledger. If not, post it.
   955→    if (reward.status === 'calculated') {
   956→      await postRewardToLedger(tenantId, { rewardId: reward.id }, `vpp-reward-${reward.id}`)
   957→      // Reload reward to get updated status.
   958→      reward = (await db.reward.findUnique({ where: { id: reward.id } }))!
   959→      await db.vppDispatchAssignment.update({ where: { id: assignmentId }, data: { economicStage: 'ledger_posted' } })
   960→    }
   961→
   962→    // 3. Check if settlement exists. If not, create it.
   963→    let settlement = await db.settlement.findUnique({ where: { rewardId: reward.id } })
   964→    if (!settlement) {
   965→      settlement = await createSettlement(tenantId, reward.id)
   966→      await db.vppDispatchAssignment.update({ where: { id: assignmentId }, data: { economicStage: 'settlement_pending' } })
   967→    }
   968→
   969→    // 4. Process the specific settlement if not yet completed.
   970→    if (settlement.status !== 'completed') {
   971→      await processSettlementForReward(tenantId, reward.id)
   972→      settlement = (await db.settlement.findUnique({ where: { rewardId: reward.id } }))!
   973→    }
   974→
   975→    // 5. Check if settlement is now completed.
   976→    if (settlement.status === 'completed') {
   977→      await db.vppDispatchAssignment.update({
   978→        where: { id: assignmentId },
   979→        data: { status: 'completed', economicStage: 'completed', completedAt: new Date() },
   980→      })
   981→      const pending = await db.vppDispatchAssignment.count({
   982→        where: { dispatchId: assignment.dispatchId, status: { not: 'completed' } },
   983→      })
   984→      if (pending === 0) {
   985→        await db.vppDispatch.update({ where: { id: assignment.dispatchId }, data: { status: 'completed' } })
   986→      }
   987→      await appendAudit({
   988→        tenantId, actorId,
   989→        eventType: 'vpp.reconciliation_completed',
   990→        resourceType: 'vpp_dispatch_assignment',
   991→        resourceId: assignmentId,
   992→        metadata: { rewardId: reward.id, settlementId: settlement.id },
   993→      })
   994→      return { assignment_id: assignmentId, status: 'completed', economic_stage: 'completed', message: 'Reconciliation succeeded' }
   995→    }
   996→
   997→    // Settlement still not completed.
   998→    await db.vppDispatchAssignment.update({ where: { id: assignmentId }, data: { status: 'reconciliation_required' } })
   999→    return {
  1000→      assignment_id: assignmentId,
  1001→      status: 'reconciliation_required',
  1002→      economic_stage: assignment.economicStage,
  1003→      message: 'Reconciliation attempted but settlement not yet completed. Manual review may be needed.',
  1004→    }
  1005→  } catch (err) {
  1006→    // Reconciliation itself failed → back to reconciliation_required.
  1007→    await db.vppDispatchAssignment.update({ where: { id: assignmentId }, data: { status: 'reconciliation_required' } })
  1008→    throw err
  1009→  }
  1010→}
