   253→   253→function toneForStatus(status: string): Tone {
   254→   254→  const s = status.toLowerCase()
   255→   255→  if (
   256→   256→    ["verified", "active", "completed", "settled", "published", "provisioned", "registered"].includes(s)
   257→   257→  )
   258→   258→    return "emerald"
   259→   259→  if (["submitted", "processing"].includes(s)) return "sky"
   260→   260→  if (["pending", "calculated", "created", "retrying", "draft"].includes(s)) return "amber"
   261→   261→  if (["rejected", "failed", "suspended", "revoked", "voided", "disputed", "offboarded"].includes(s))
   262→   262→    return "rose"
   263→   263→  if (["posted", "archived"].includes(s)) return "teal"
   264→   264→  return "muted"
   265→   265→}

   436→   436→function RelativeTime({ date }: { date: string | Date | null | undefined }) {
   437→   437→  if (!date) return <span className="text-muted-foreground">—</span>

   449→   449→      suppressHydrationWarning

   460→   460→function useApi<T>(
   461→   461→  path: string | null,
   462→   462→  tenantId: string | undefined,
   463→   463→  refreshKey: number,
   464→   464→): { data: T | null; loading: boolean; error: string | null } {
   465→   465→  const currentKey = `${path ?? ""}::${tenantId ?? ""}::${refreshKey}`
   466→   466→  const [state, setState] = useState<{
   467→   467→    data: T | null
   468→   468→    loading: boolean
   469→   469→    error: string | null
   470→   470→    key: string
   471→   471→  }>({ data: null, loading: false, error: null, key: "" })
   472→   472→
   473→   473→  useEffect(() => {
   474→   474→    if (!path) return
   475→   475→    let cancelled = false
   476→   476→    // Defer the "loading" transition out of the effect body (avoids a synchronous
   477→   477→    // setState-in-effect). The request itself is fired immediately.
   478→   478→    queueMicrotask(() => {
   479→   479→      if (cancelled) return
   480→   480→      setState((s) => ({ data: s.data, loading: true, error: null, key: currentKey }))
   481→   481→    })
   482→   482→    apiFetch<T>(path, { tenantId })
   483→   483→      .then((d) => {
   484→   484→        if (cancelled) return
   485→   485→        setState({ data: d, loading: false, error: null, key: currentKey })
   486→   486→      })
   487→   487→      .catch((e: unknown) => {
   488→   488→        if (cancelled) return
   489→   489→        setState({
   490→   490→          data: null,
   491→   491→          loading: false,
   492→   492→          error: e instanceof Error ? e.message : "Unknown error",
   493→   493→          key: currentKey,
   494→   494→        })
   495→   495→      })
   496→   496→    return () => {
   497→   497→      cancelled = true
   498→   498→    }
   499→   499→  }, [currentKey, path, tenantId])
   500→   500→
   501→   501→  // Loading when the requested key differs from the last-completed key.
   502→   502→  const loading = state.key !== currentKey || state.loading
   503→   503→  return { data: state.data, loading, error: state.error }
   504→   504→}
